# Generates the self-made test models in Assets/Models with the installed Autodesk FBX SDK.
# Nothing is downloaded; the SDK license is accepted by the user when installing it.
cmake_minimum_required(VERSION 3.24)
project(FbxSampleGen LANGUAGES CXX)
set(DXF_FBX_SDK_ROOT "" CACHE PATH "Installed Autodesk FBX SDK root (contains include/fbxsdk.h)")
if(NOT EXISTS "${DXF_FBX_SDK_ROOT}/include/fbxsdk.h")
    message(FATAL_ERROR "Set DXF_FBX_SDK_ROOT to the installed FBX SDK")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(FbxSampleGen Main.cpp)
target_compile_features(FbxSampleGen PRIVATE cxx_std_20)
target_compile_options(FbxSampleGen PRIVATE /W4 /utf-8 /EHsc)
target_include_directories(FbxSampleGen PRIVATE "${DXF_FBX_SDK_ROOT}/include")
foreach(_dxf_config IN ITEMS Debug Release)
    string(TOLOWER "${_dxf_config}" _dxf_sub)
    set(_dxf_dir "${DXF_FBX_SDK_ROOT}/lib/x64/${_dxf_sub}")
    if(NOT EXISTS "${_dxf_dir}/libfbxsdk-mt.lib")
        set(_dxf_dir "${DXF_FBX_SDK_ROOT}/lib/vs2022/x64/${_dxf_sub}")
    endif()
    foreach(_dxf_name IN ITEMS libfbxsdk-mt libxml2-mt zlib-mt)
        if(EXISTS "${_dxf_dir}/${_dxf_name}.lib")
            target_link_libraries(FbxSampleGen PRIVATE "$<$<CONFIG:${_dxf_config}>:${_dxf_dir}/${_dxf_name}.lib>")
        endif()
    endforeach()
endforeach()
# libxml2 2.15 in the FBX SDK uses BCryptGenRandom (Windows system library).
target_link_libraries(FbxSampleGen PRIVATE bcrypt)
