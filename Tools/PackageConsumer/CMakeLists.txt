# 再配置したパッケージをfind_packageで使う外部の利用者。ソースツリーのPrivateヘッダーや元のBuildの生成物は参照しない。
cmake_minimum_required(VERSION 3.24)
project(RelocatedConsumer LANGUAGES CXX)
option(DXF_VALIDATE_NATIVE "Build the external application that uses dxf::native" OFF)
find_package(dxlib_framework CONFIG REQUIRED)
if(NOT TARGET dxf::toolbox OR NOT TARGET dxf::foundation OR NOT TARGET dxf::support OR NOT TARGET dxf::runtime OR NOT TARGET dxf::gameplay)
    message(FATAL_ERROR "Layer targets are missing from the installed package")
endif()
# Physicsだけの利用者へ、Native・Debug・描画の依存が逆流していないこと。
get_target_property(_dxf_physics_links dxf::physics INTERFACE_LINK_LIBRARIES)
if(_dxf_physics_links MATCHES "native|debug|DxLib|support|runtime|gameplay")
    message(FATAL_ERROR "dxf::physics depends on a higher layer: ${_dxf_physics_links}")
endif()
add_executable(Consumer Main.cpp)
target_link_libraries(Consumer PRIVATE dxf::framework dxf::debug_tools)
add_executable(PhysicsOnly Physics.cpp)
target_link_libraries(PhysicsOnly PRIVATE dxf::physics)
add_executable(SupportOnly Support.cpp)
target_link_libraries(SupportOnly PRIVATE dxf::support)
if(DXF_VALIDATE_NATIVE)
    if(NOT TARGET dxf::native)
        message(FATAL_ERROR "The installed package has no dxf::native")
    endif()
    add_executable(NativeApp NativeApp.cpp)
    target_link_libraries(NativeApp PRIVATE dxf::native)
    target_compile_definitions(NativeApp PRIVATE DX_NON_USING_NAMESPACE_DXLIB NOMINMAX)
    # DxLibのVC SDK（とソースからのビルド）は静的CRT。パッケージが用意する関数で合わせる。
    foreach(_dxf_target Consumer PhysicsOnly SupportOnly NativeApp)
        dxf_use_static_runtime(${_dxf_target})
    endforeach()
endif()
